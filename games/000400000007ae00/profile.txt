# New Super Mario Bros. 2 (USA, 000400000007ae00)

# Current scene: the profile id of the running scene (the scene manager's "current" field; the u16 just before
# it, 0x5ee930, is the pending request and reads 0xffff between transitions).
#   4 StartUp (first-boot / system dialogs), 5 TitleMenu (title, solo/co-op, file select, options),
#   6 OpeningDemo, 16 Course (a level), 17 CourseIn ("World 1-1 / lives" card), 18 CourseSelect (world map),
#   19 GameOver; 7-15 Coin Rush, shop, StreetPass and error scenes.
value scene = u16 0x5ee932

# The play scene's game info (a heap object; the global pointer is null outside levels).
value course = u32 0x5e61e8
value lives = s32 0x5e61e8 > 0x80
value coins = s32 0x5e61e8 > 0x8c          # 0-99 coin counter (already on the top screen)
value score = s32 0x5e61e8 > 0xb4          # the score shown on the bottom screen
value course_coins = s32 0x5e61e8 > 0x178  # coins collected in this attempt (bottom screen, next to "Best")
value star0 = s8 0x5e61e8 > 0xb8           # star coins of this level: -1 not collected, else the player (0/1)
value star1 = s8 0x5e61e8 > 0xb9
value star2 = s8 0x5e61e8 > 0xba
value item = u8 0x5e61e8 > 0xdf            # stored item, player 1: 9 none, 0 mushroom, 1 fire flower, 2 star,
                                           # 4 mini mushroom, 6 leaf, 7 gold flower, 8 white leaf
# The same game info kept between levels (copied back when a level ends; stale during a level).
value saved_lives = s32 0x5e6130 > 0x34
value saved_item = u8 0x5e6130 > 0x93
