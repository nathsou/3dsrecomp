# Animal Crossing: New Leaf – Welcome amiibo (EUR). Format: runtime/profile.cpp.
# The USA edition's profile (games/0004000000198e00) with this build's addresses: the CRO modules are the same, the
# program's rodata is 0x1180 lower here, its data and bss 0x1000 lower (0x1004 from the keyboard's instance on), and
# the save's layout is the same.

# The game's own keyboard (names, the town's name, letters, chat): BsSkb, whose instance is set while it is up.
value skb = u32 0x94f644
keyboard when skb != 0

# the keyboard's "ABC" page, bottom-screen pixels (the European one: ' = / and ? ! @ where the American has : [ ]
# and ' /)
key 1234567890- at 15 78 step 25
key qwertyuiop at 18 102 step 25
key asdfghjkl'=/ at 23 123 step 25
key zxcvbnm,.?!@ at 27 146 step 25
key backspace at 295 78
key shift at 85 167
key space at 162 167
key enter at 260 223           # Confirm (chat: send)
key escape at 40 223           # Close (chat; the name keyboards have none)

# Resetti: each player's "session under way" bit (set when play starts, cleared by both saves), in the first part of
# the player's data, a CRC-32C block (players 0xa480 apart). A save made on a console with the base game and its
# Welcome amiibo update (0004000000086400) has the same garden_plus.dat.
unsaved garden_plus.dat 0x57aa 0x02 crc32c 0xa0 0x6b84
unsaved garden_plus.dat 0xfc2a 0x02 crc32c 0xa520 0x6b84
unsaved garden_plus.dat 0x1a0aa 0x02 crc32c 0x149a0 0x6b84
unsaved garden_plus.dat 0x2452a 0x02 crc32c 0x1ee20 0x6b84

# ---- one screen (the "Top screen only" layout): the bottom screen never shows as such. Its menus come up as a
# panel over the top screen, its background (the plaid wallpaper) hidden so dialogs float; the field's map and
# tabs are reached through the hub (ZL), the map (-) and the other overlays.
onescreen
value single = runtime single
value debug = runtime debug
value map = u32 0x94ec3c > 0                   # the BsMenuMap page's vtable (0x8f814c) while the field's map shows:
                                                # the player is free in the field (dialogs hide it)
# BsMenuBg's layout: N_mvmnt_Uscrn > P_base_00 (the plaid), hidden (alpha 0: not drawn). The bottom screen's pass
# reuses the top screen's render target, so it is cleared to transparent first ("clearbottom").
clearbottom
value bg_alpha = u8 0x94e708 > 0x78 > 0xbc > 0x14 > 0xb0
set bg_alpha = 0 when single == 1
set bg_alpha = 255 when single == 0 and debug == 0
set bg_alpha = 0 when debug == 1
# The title screen's bottom layout (BsTitle, found through the game's tree of modes, slot 0x3fc): N_All > N_set
# holds only its background pictures. Guarded: the slot must hold a BsTitle and the pane be named "N_set". (The
# tree's nodes are made in order: booting without a save puts BsTitle elsewhere, at 0x6b8, and the plaid stays.)
value title = u32 0x97428c > 0x3fc > 0                            # the slot's object's vtable
value title_set_name = u32 0x97428c > 0x3fc > 0x620 > 0x14 > 0x10 > 0xb4
value title_set_flags = u8 0x97428c > 0x3fc > 0x620 > 0x14 > 0x10 > 0xb3
set title_set_flags = 0 when single == 1 and title == 0x40760e18 and title_set_name == 0x65735f4e
set title_set_flags = 1 when single == 0 and debug == 0 and title == 0x40760e18 and title_set_name == 0x65735f4e
focus top when map == 0x8f814c
focus panel
value pockets_open = u32 0x94e9dc > 0          # the BsMenuItem page's vtable (0x8e9d94) while the pockets show
                                                # (the pointer stays after they close; the page's memory is reused)
value pocket0 = u32 0xaae14c > 0x6bd0           # the current player's first pocket (16 items, 4 bytes apart)
value held = u16 0xaae14c > 0x26                # the item in the player's hands

# Controls in the manner of a single-screen Animal Crossing: X the pockets (the game's own), ZL a hub of the bottom
# screen's buttons, - the map, the d-pad the tools in the pockets (up: the ring; left/right: the previous/next one;
# down: put it away), the right stick the camera (up: look up, as L and R do) and a pointer over menus (ZR
# touches where it points).
bind zl hub
bind back map
bind dup tools
bind dleft tool-prev
bind dright tool-next
bind ddown tool-away
bind zr pointer
bind rs-up R

# the hub: the field's buttons (bottom-screen pixels; outdoors only: indoors the bottom screen has none), their
# icons kept from the bottom screen
hub "Pockets" press X icon 2 0 46 22 when held >= 0
hub "Designs" tap 72 12 icon 54 0 46 22 when map == 0x8f814c
hub "Critterpedia" tap 123 12 icon 106 0 48 22 when map == 0x8f814c
hub "TPC" tap 178 12 icon 158 0 48 22 when map == 0x8f814c
hub "Map" map when map == 0x8f814c
hub "Tools" tools when held >= 0
hub "Chat" tap 160 226 icon 136 214 48 26 when map == 0x8f814c
hub "Save" press START glyph "✓" when held >= 0
map 64 27 192 184 when map == 0x8f814c

# the tool ring: pockets whose category is a tool (axes to toy hammers), held with the game's own request (the one
# the pockets' "Hold" makes: it finds the player and starts the change of tools), made as the player's update
# (AcPlayer's vtable slot 9) is called
tools pockets pocket0 held held empty 0x7ffe
tools items /Item/Param/Item.bin record 30 first 0x2000 category 10 kinds 0x0e 0x14
tools icon 30
tools names /Script/Str/STR_Item_name.umsbt
tools slots 35 115 85 115 235 115 285 115 35 157 85 157 135 157 185 157 235 157 285 157 35 197 85 197 135 197 185 197 235 197 285 197
tools equip 160 80 open X when-open pockets_open == 0x8e9d94 ready map == 0x8f814c
tools hold 0x5c0bc4 on 0x68c0ec
