# Animal Crossing: New Leaf – Welcome amiibo, every region (USA 0004000000198e00, EUR 0004000000198f00: their
# games/<title id>/profile.txt link here). Format: runtime/profile.cpp.
#
# No address is written down: each one is found in the running game through its C++ classes (RTTI), so the same
# profile serves every build. The log lists what each symbol came to.

# the classes whose objects the profile recognises (their vtable pointers)
symbol BsMenuMap = vtable BsMenuMap             # the field's map page on the bottom screen
symbol BsMenuItem = vtable BsMenuItem           # the pockets
symbol BsMenuItemSelect = vtable BsMenuItemSelect   # the pockets as a shop's counter shows them (what to sell)
symbol BsTitle = vtable BsTitle                 # the title screen (in a CRO module)
# globals and functions, reached from the classes' code
symbol skb_instance = vtable BsSkb slot 3 global 2          # the keyboard's instance (set while it is up)
symbol map_page = vtable BsMenuMap user 4 global 0          # its factory stores the map page there
symbol menu_bg = vtable BsMenuBg slot 3 global 0            # the menus' background
symbol pockets_page = vtable BsMenuItem slot 6 global 0     # the pockets page (BsMenuItem's static data follows)
symbol mode_tree = vtable BsRoot slot 8 call 0 global 0     # the root of the game's tree of modes
symbol hold_handler = pockets_page + 0x30 > 0               # the pockets' "Hold" (in BsMenuItem's table of states)
symbol hold_request = hold_handler call 3                   # the request it makes: finds the player, changes tools
symbol player = hold_handler call 0 global 0                # the current player's data
symbol player_update = vtable AcPlayer slot 9               # AcPlayer's update
symbol player_actor = hold_request call 0 global 0 + 0x29204   # the table its lookup reads: the player's AcPlayer
symbol menu_pages = vtable BsMenuItem slot 7 call 0 global 0 + 4  # the open pockets-like page's holder (its +0xc)

# The game's own keyboard (names, the town's name, letters, chat): BsSkb, whose instance is set while it is up.
value skb = u32 skb_instance
keyboard when skb != 0

key backspace at 295 78
key shift at 85 167
key space at 162 167

key escape at 40 223           # Close (chat; the name keyboards have none)

# Resetti: each player's "session under way" bit (set when play starts, cleared by both saves), in the first part of
# the player's data, a CRC-32C block (players 0xa480 apart). Every region's save has this layout, and so does a save
# from a console with the base game and its Welcome amiibo update (0004000000086400).
unsaved garden_plus.dat 0x57aa 0x02 crc32c 0xa0 0x6b84
unsaved garden_plus.dat 0xfc2a 0x02 crc32c 0xa520 0x6b84
unsaved garden_plus.dat 0x1a0aa 0x02 crc32c 0x149a0 0x6b84
unsaved garden_plus.dat 0x2452a 0x02 crc32c 0x1ee20 0x6b84

# ---- one screen (the "Top screen only" layout): the bottom screen never shows as such. Its menus come up as a
# panel over the top screen, its background (the plaid wallpaper) hidden so dialogs float; the field's map and
# tabs are reached through the hub (ZL), the map (-) and the other overlays.
onescreen
value single = runtime single
value debug = runtime debug
value actor = u32 player_actor                  # the player's AcPlayer (none on the title screen, nor on the train
                                                # before a new game's town is chosen)
value map = u32 map_page > 0                   # BsMenuMap while the field's map shows: the player is free in the
                                                # field (dialogs hide it)
# BsMenuBg's layout: N_mvmnt_Uscrn > P_base_00 (the plaid), hidden (alpha 0: not drawn). The bottom screen's pass
# reuses the top screen's render target, so it is cleared to transparent first ("clearbottom").
clearbottom
value bg_alpha = u8 menu_bg > 0x78 > 0xbc > 0x14 > 0xb0
# (kept while talking: a page that opens then, such as a shop's "Sell what?", shows as the 3DS shows it, a card over
# the dialog, its bubbles' dotted shade on the plaid it is meant for)
value talk_state = u8 player_actor > 0x1a9
set bg_alpha = 0 when single == 1 and talk_state != 0x0f
set bg_alpha = 255 when single == 1 and talk_state == 0x0f
set bg_alpha = 255 when single == 0 and debug == 0
set bg_alpha = 0 when debug == 1
# The title screen's bottom layout (BsTitle, found through the game's tree of modes, slot 0x3fc): N_All > N_set
# holds only its background pictures. Guarded: the slot must hold a BsTitle and the pane be named "N_set". (The
# tree's nodes are made in order: booting without a save puts BsTitle elsewhere, at 0x6b8, and the plaid stays.)
value title = u32 mode_tree > 0x3fc > 0                           # the slot's object's vtable
value title_set_name = u32 mode_tree > 0x3fc > 0x620 > 0x14 > 0x10 > 0xb4
value title_set_flags = u8 mode_tree > 0x3fc > 0x620 > 0x14 > 0x10 > 0xb3
set title_set_flags = 0 when single == 1 and title == BsTitle and title_set_name == 0x65735f4e
set title_set_flags = 1 when single == 0 and debug == 0 and title == BsTitle and title_set_name == 0x65735f4e
# The bottom screen shows as a panel only when it has something to touch: the pockets and the pages like them (a
# shop's "Sell what?"), the keyboard, a dialog's choices, menus. Standing, walking or talking (AcPlayer's state), it shows only its tabs
# or a dimmed pattern behind the dialog: the top screen alone.
# No player yet (the title screen, between modes): the top screen alone too, the title's "Press A" and copyright
# lines would cover its logo.
value player_state = u8 player_actor > 0x1a9
value item_page = u32 menu_pages > 0xc          # the open pockets-like page (none: 0)
value bottom_dark = runtime bottom_dark          # how much of the bottom screen is dark (percent)
focus top when map == BsMenuMap and actor != 0
focus panel when map == BsMenuMap               # the map without a player: a new game's choice of town (touched)
focus panel when skb != 0
focus panel when item_page != 0
focus top when actor == 0
focus top when player_state == 0x16            # opening the pockets, before their page is up
# talking, a list of choices up (a dark box on the plaid): once its box has opened (it shows a stale picture for
# about 30 frames as it does)
focus panel after 32 when player_state == 0x0f and bottom_dark > 6
focus top when player_state == 0x06 or player_state == 0x0d or player_state == 0x0f
focus panel
value pockets_open = u32 pockets_page > 0      # BsMenuItem while the pockets show (the pointer stays after they
                                                # close; the page's memory is reused)
value pocket0 = u32 player > 0x6bd0             # the current player's first pocket (16 items, 4 bytes apart)
value held = u16 player > 0x26                  # the item in the player's hands

# Controls in the manner of a single-screen Animal Crossing: X the pockets (the game's own), ZL a hub of the bottom
# screen's buttons, - the map, the d-pad the tools in the pockets (up: the ring; left/right: the previous/next one;
# down: put it away), the right stick the camera (up: look up, as L and R do) and a pointer over menus (ZR
# touches where it points).
bind zl hub
bind back map
bind dup tools
bind dleft tool-prev
bind dright tool-next
bind ddown tool-away
bind zr pointer
bind rs-up R

# stacks in the pockets (the menu's "Stack items", on by default), as in New Horizons and with its sizes: seed bags,
# saplings, shrub starts, shells, mushrooms, flowers, pitfall seeds, fertilizer, wrapping paper, candy, lollipops,
# fortune cookies, coffee beans, cooking ingredients, medicine and chocolates up to 10, nuggets and gems up to 30.
# Fruit, stationery, turnips and Bells count in the game already; tools, clothes, furniture, fish, bugs, fossils
# and the like do not stack in New Horizons either. Each player of each town
# (ids next to their names in the player data) keeps their own; the pockets are tidied as the player's update runs,
# while the player stands or walks (not in a dialog, a menu or a shop's screen). As a conversation starts (a shop's
# counter, a villager) the stacks are laid out in free pockets, one left free for what it may give, so a whole stack
# sells at once; the counts show on the shops' sell screen too (BsMenuItemSelect); with the pockets full, the stack
# of the item the player stands on makes room for it.
value player_id = u16 player > 0x55a6
value town_id = u16 player > 0x55bc
stack owner town_id player_id
stack on player_update
stack free 0x1a9 0x06 0x0d                     # AcPlayer's state: standing, walking (indoors too)
stack spread 0x0f keep 1                        # ... talking (a shop's counter, a villager)
stack underfoot 0x994                           # AcPlayer's item underfoot (its name shows over the player)
value open_page = u32 menu_pages > 0xc > 0
stack shown when open_page == BsMenuItemSelect
stack kinds 0x36 0x37 0x38 0x3c 0x3e 0x40 0x41 0x46 0x49 0x52 0x53 0x5a 0x71 max 10
stack kinds 0x42 0x43 max 30
stack kinds 0x44 0x4a 0x72 0x87 max 10            # cooking ingredients, medicine, chocolates
